import json, ssl, time, urllib.error, urllib.request

        return None

    line_id,base=line

        context=ssl.create_default_context()
        opener=urllib.request.build_opener(urllib.request.HTTPSHandler(context=context),NoRedirect())
        req=urllib.request.Request(base+'/api/status',headers={'User-Agent':'AICLI-Client-A5-01-status-probe'})
        started=time.perf_counter()
        row={'line':line_id,'attempt':attempt,'at':datetime.now(timezone.utc).isoformat(),'tls':'not_completed','http_status':None}
        try:
            with opener.open(req,timeout=15) as response:
                response.read(1)
                row.update(tls='verified',http_status=response.status)
        except urllib.error.HTTPError as error:
            row.update(tls='verified',http_status=error.code,error='http_status')

            row['error']=type(error.reason).__name__

            row['error']=type(error).__name__
        row['latency_ms']=round((time.perf_counter()-started)*1000,2)
        rows.append(row)
    return rows
